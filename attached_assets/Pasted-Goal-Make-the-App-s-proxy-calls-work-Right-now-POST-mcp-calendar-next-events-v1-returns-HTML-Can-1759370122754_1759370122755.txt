// src/server.ts
import express from "express";
import type { Request, Response } from "express";
import { authMiddleware } from "./auth";
import { toolHandlers } from "./tools"; // map: { "calendar.next_events.v1": handler, ... }

const router = express.Router();

router.get("/contracts", (_req, res) => {
  res.json({ tools: Object.keys(toolHandlers) });
});

router.post("/mcp/:tool", authMiddleware, async (req: Request, res: Response) => {
  const rid = String(req.header("x-request-id") || crypto.randomUUID());
  const key = req.params.tool;
  const handler = (toolHandlers as any)[key];
  const t0 = Date.now();
  res.setHeader("x-request-id", rid);
  if (!handler) return res.status(404).json({ error: { code: "UNKNOWN_TOOL", message: key } });
  try {
    const result = await handler(req, res); // if your handlers write directly, just `return`
    if (!res.headersSent) res.json(result);
  } catch (e: any) {
    console.error("[MCP]", { rid, route: `/mcp/${key}`, err: e?.message });
    if (!res.headersSent) res.status(500).json({ error: { code: "TOOL_ERROR", message: e?.message || "Internal Error" } });
  } finally {
    console.log("[MCP]", { rid, route: `/mcp/${key}`, status: res.statusCode, ms: Date.now() - t0 });
  }
});

// ensure POST /agent/act is mounted as before
app.use(router);
