const mem = new Map<string, Tokens>(); // dev; swap to DB later

export async function saveGoogleTokens(userId: string, t: Tokens) {
  mem.set(userId, t);
}
export async function getGoogleTokens(userId: string) {
  return mem.get(userId) || null;
}
export async function clearGoogleTokens(userId: string) {
  mem.delete(userId);

D) Google OAuth routes

Install googleapis: pnpm add googleapis

Add src/lib/google.ts:

import { google } from "googleapis";
const { GOOGLE_CLIENT_ID, GOOGLE_CLIENT_SECRET, GOOGLE_OAUTH_REDIRECT_URI } = process.env;

export function getOAuthClient() {
  if (!GOOGLE_CLIENT_ID || !GOOGLE_CLIENT_SECRET || !GOOGLE_OAUTH_REDIRECT_URI)
    throw new Error("Google OAuth env not set");
  return new google.auth.OAuth2(GOOGLE_CLIENT_ID, GOOGLE_CLIENT_SECRET, GOOGLE_OAUTH_REDIRECT_URI);
}

export async function makeClientsFor(userId: string, tokens: any) {
  const oauth2 = getOAuthClient();
  oauth2.setCredentials(tokens);
  return {
    calendar: google.calendar({ version: "v3", auth: oauth2 }),
    gmail: google.gmail({ version: "v1", auth: oauth2 }),
  };
}


Add routes in src/server.ts:

import { getOAuthClient, makeClientsFor } from "./lib/google";
import { saveGoogleTokens, getGoogleTokens, clearGoogleTokens } from "./lib/tokenStore";

// naive dev auth to resolve userId: from query or header (later replace with your app session)
function requireUserId(req, res, next) {
  const userId = req.query.userId || req.header("x-user-id");
  if (!userId) return res.status(400).json({ error: "Missing userId" });
  (req as any).userId = String(userId);
  next();
}

app.get("/auth/google", requireUserId, async (req, res) => {
  const userId = (req as any).userId;
  const oauth2 = getOAuthClient();
  const url = oauth2.generateAuthUrl({
    access_type: "offline",
    prompt: "consent",
    scope: [
      "https://www.googleapis.com/auth/calendar.readonly",
      "https://www.googleapis.com/auth/gmail.readonly",
    ],
    state: encodeURIComponent(JSON.stringify({ userId })),
  });
  res.redirect(url);
});

app.get("/auth/google/callback", async (req, res) => {
  const { code, state } = req.query as { code?: string; state?: string };
  if (!code) return res.status(400).send("Missing code");
  const { userId } = JSON.parse(decodeURIComponent(state || "{}"));
  const oauth2 = getOAuthClient();
  const { tokens } = await oauth2.getToken(code);
  await saveGoogleTokens(userId, {
    accessToken: tokens.access_token!,
    refreshToken: tokens.refresh_token!,
    expiryDate: tokens.expiry_date,
  });
  res.status(200).send("Google connected. You can close this tab.");
});

app.post("/auth/google/disconnect", requireUserId, async (req, res) => {
  await clearGoogleTokens((req as any).userId);
  res.json({ ok: true });
});

app.get("/me/google", requireUserId, async (req, res) => {
  const t = await getGoogleTokens((req as any).userId);
  res.json({ connected: !!t });
});

E) Make tools use stored tokens

Update these handlers to fetch tokens and call real APIs:

Example (calendar.next_events.v1):

import { devToolAuth } from "./middleware/devToolAuth";
import { getGoogleTokens } from "./lib/tokenStore";
import { makeClientsFor } from "./lib/google";

app.post("/mcp/calendar.next_events.v1", devToolAuth, async (req, res) => {
  const { userId, timeMin, timeMax, maxResults = 50 } = req.body || {};
  if (!userId) return res.status(400).json({ error: "Missing userId" });

  const tokens = await getGoogleTokens(userId);
  if (!tokens) return res.status(401).json({ error: { code: "GOOGLE_NOT_CONNECTED", message: "Connect Google first" }});

  const { calendar } = await makeClientsFor(userId, {
    access_token: tokens.accessToken,
    refresh_token: tokens.refreshToken,
    expiry_date: tokens.expiryDate,
  });

  const { data } = await calendar.events.list({
    calendarId: "primary",
    timeMin, timeMax, singleEvents: true, orderBy: "startTime", maxResults
  });

  const events = (data.items || []).map(e => ({
    id: e.id, summary: e.summary,
    attendees: (e.attendees || []).map(a => a.email).filter(Boolean),
    start: e.start?.dateTime || e.start?.date,
    end: e.end?.dateTime || e.end?.date,
    hangoutLink: e.hangoutLink || null,
  }));

  res.json({ events });
});
