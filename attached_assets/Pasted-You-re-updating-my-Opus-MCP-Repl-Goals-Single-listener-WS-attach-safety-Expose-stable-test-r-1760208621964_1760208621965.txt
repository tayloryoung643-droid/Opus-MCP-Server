import http from "http";

B. Tool dev-auth (bearer) for local tests

For every route under /mcp/* (your tool HTTP endpoints), add this minimal dev gate before the handler body:

// middleware/devToolAuth.ts
import { NextFunction, Request, Response } from "express";
export function devToolAuth(req: Request, res: Response, next: NextFunction) {
  const devKey = process.env.DEV_LOCAL_TOOL_KEY;
  if (!devKey) return res.status(500).json({ error: "DEV_LOCAL_TOOL_KEY not set" });
  const auth = req.header("authorization") || "";
  if (auth !== `Bearer ${devKey}`) return res.status(401).json({ error: "Unauthorized" });
  next();
}

// in server.ts or your tools router
import { devToolAuth } from "./middleware/devToolAuth";

// Example tool route
app.post("/mcp/calendar.next_events.v1", devToolAuth, async (req, res) => {
  // existing implementation...
});

C. Print the public URL clearly

Above already logs process.env.REPLIT_URL if present. If not present, derive it once on startup:

// if REPLIT_URL is not injected, try to infer from X-Forwarded-Host on first request:
app.use((req, _res, next) => {
  if (!process.env.REPLIT_URL && req.headers["x-forwarded-host"]) {
    process.env.REPLIT_URL = `https://${req.headers["x-forwarded-host"]}`;
    console.log(`[config] Detected public URL: ${process.env.REPLIT_URL}`);
  }
  next();
});

D. One-command smoke test

Add scripts/smoke.sh:

#!/usr/bin/env bash
set -euo pipefail

HOST="${1:-${REPLIT_URL:-}}"
if [[ -z "${HOST}" ]]; then
  echo "Usage: ./scripts/smoke.sh https://<your-public-host>"
  exit 1
fi

echo "→ Health"
curl -fsS "$HOST/healthz" | jq .

echo "→ Contracts"
curl -fsS "$HOST/contracts" | jq .

echo "→ Tool (dev auth) calendar.next_events.v1"
curl -fsS -X POST "$HOST/mcp/calendar.next_events.v1" \
  -H "Authorization: Bearer ${DEV_LOCAL_TOOL_KEY}" \
  -H "Content-Type: application/json" \
  --data '{"userId":"test-user","timeMin":"2025-10-11T00:00:00Z","timeMax":"2025-10-18T00:00:00Z"}' | jq .
